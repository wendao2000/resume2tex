% Measure through public content hooks, then reuse the typeset material.
\newcommand{\ResumeSectionMeasureBegin}{%
  \setbox\ResumeSectionMeasureBox=\vbox\bgroup%
}
\newcommand{\ResumeSectionMeasureEnd}[1]{%
  \par\unskip\egroup%
  % Measurements must survive tcolorbox's title and body groups.
  % A page-sized value suffices for oversized content; avoid dimension overflow.
  \global#1=\textheight%
  \ifnum\ht\ResumeSectionMeasureBox<\number\textheight%
    \ifnum\ht\ResumeSectionMeasureBox<0 \else%
      \global#1=\dimexpr\ht\ResumeSectionMeasureBox+\dp\ResumeSectionMeasureBox\relax%
    \fi%
  \fi%
  \unvbox\ResumeSectionMeasureBox%
}
\newcommand{\ResumeSectionBefore}{%
  \par\addvspace{0.6em}%
  % All other frame, separator, and padding dimensions below are zero.
  \setlength{\ResumeSectionHeight}{\dimexpr\ResumeSectionBodyHeight+\ResumeSectionTitleHeight+\ResumeSectionTopSep+2pt\relax}%
  \ifdim\ResumeSectionHeight>\textheight%
    \Needspace{4\baselineskip}%
  \else%
    \Needspace{\ResumeSectionHeight}%
  \fi%
  \noindent%
}
\newtcolorbox{resumesection}[1]{%
  breakable,colframe=primary,colback=white,colbacktitle=white,coltitle=primary,
  extras={frame empty,interior empty,interior titled empty},
  fonttitle=\large\bfseries,
  title={#1\par\vspace{7pt}\hrule height 0.4pt},
  title after break={#1 (cont.)\par\vspace{7pt}\hrule height 0.4pt},
  boxrule=0pt,titlerule=0pt,arc=0pt,outer arc=0pt,
  boxsep=0pt,left=0pt,right=0pt,top=\ResumeSectionTopSep,bottom=0pt,
  lefttitle=0pt,righttitle=0pt,toptitle=0pt,bottomtitle=0pt,
  before=\ResumeSectionBefore,after={\par},
  before title=\ResumeSectionMeasureBegin,
  after title={\ResumeSectionMeasureEnd{\ResumeSectionTitleHeight}},
  before upper={\setlength{\parindent}{0pt}\setlength{\parskip}{0.25em}\ResumeSectionMeasureBegin},
  after upper={\ResumeSectionMeasureEnd{\ResumeSectionBodyHeight}},
  pad before break=0pt,pad after break=\ResumeSectionTopSep,lines before break=3
}
